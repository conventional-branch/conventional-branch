{{- /* Renders the AI agent prefix registry (data/agents.yaml) as a table. Column headers and the generic-agent label fall back to English. An entry with alias_of is marked as an alias of that prefix. An archived version's page lists only the entries that version had (since <= it). */ -}} {{- $data := .Site.Data.agents -}} {{- $pageVersion := partial "version-current.html" .Page -}} {{- $versions := slice -}} {{- $reached := false -}} {{- range site.Params.versions.list -}} {{- if not $reached -}}{{- $versions = $versions | append . -}}{{- end -}} {{- if eq . $pageVersion -}}{{- $reached = true -}}{{- end -}} {{- end -}} {{- $lang := .Page.Lang | default "en" -}} {{- $labels := $data.labels -}} {{- $hPrefix := index $labels.prefix $lang | default $labels.prefix.en -}} {{- $hAgent := index $labels.agent $lang | default $labels.agent.en -}} {{- $hVendor := index $labels.vendor $lang | default $labels.vendor.en -}} {{- $hSince := index $labels.since $lang | default $labels.since.en -}} {{- range $data.agents -}} {{- if in $versions (printf "v%s" .since) -}} {{- $name := .name -}} {{- if .generic -}} {{- $name = index $labels.generic $lang | default $labels.generic.en -}} {{- end -}} {{- end -}} {{- end -}}
{{ $hPrefix }} {{ $hAgent }} {{ $hVendor }} {{ $hSince }}
{{ .prefix }}/ {{ with .homepage }}{{ $name }}{{ else }}{{ $name }}{{ end }} {{- with .alias_of }} ({{ printf (index $labels.alias_of $lang | default $labels.alias_of.en) (printf "%s/" .) | safeHTML }}){{ end }} {{ .vendor }} v{{ .since }}
